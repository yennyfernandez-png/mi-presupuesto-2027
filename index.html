function doPost(e) { try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var contents = JSON.parse(e.postData.contents); // Recibe los datos de la web var data = sheet.getDataRange().getValues(); // Mapeo de filas según ID para actualizar PPTO 2027 (Columna 11 / K) for (var i = 1; i < data.length; i++) { var filaId = i; // ID de la fila en la web if (contents[filaId] !== undefined) { sheet.getRange(i + 1, 11).setValue(contents[filaId]); // Columna K (11) = PPTO 2027 } } return ContentService.createTextOutput(JSON.stringify({"result": "success"})) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({"result": "error", "error": err.message})) .setMimeType(ContentService.MimeType.JSON); } }
